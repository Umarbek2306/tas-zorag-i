function login() { fetch("/login", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ login: document.getElementById("login").value, password: document.getElementById("password").value }) }) .then(res => res.json()) .then(data => { if (data.success) { window.location.href = "/index.html"; } else { alert("Ошибка"); } }); }
